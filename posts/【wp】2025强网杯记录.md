```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
MMXEncode2024 解密函数
根据IDA Pro反编译的加密算法实现对应的解密逻辑
"""

import struct
import os
import sys


def decrypt_data(encrypted_data, key):
    """
    解密数据

    Args:
        encrypted_data: 加密的数据
        key: 解密密钥 (8字节)

    Returns:
        解密后的数据
    """
    if len(key) != 8:
        raise ValueError("密钥长度必须为8字节")

    # 仅解密完整的8字节块；尾部不足8字节按原样保留（加密端未处理尾部）
    data_len = len(encrypted_data)
    full_len = (data_len // 8) * 8

    decrypted_data = bytearray()

    # 按8字节块处理
    for i in range(0, full_len, 8):
        block = encrypted_data[i : i + 8]

        # 解密单个8字节块
        decrypted_block = decrypt_block(block, key)
        decrypted_data.extend(decrypted_block)

    # 追加尾部未加密的剩余字节
    if full_len < data_len:
        decrypted_data.extend(encrypted_data[full_len:])

    return bytes(decrypted_data)


def decrypt_block(encrypted_block, key):
    """
    解密单个8字节块

    # 根据IDA Pro分析的加密逻辑，逆向操作：
    # 加密过程：
    # 1. 与密钥XOR: v17 = _m_pxor(data, key)
    # 2. 字节交换: v18 = _m_por(_m_psrlwi(v17, 8), _m_psllwi(v17, 8))
    # 3. 循环左移1位: _m_por(_m_psllqi(v18, 1), _m_psrlqi(v18, 63))
    # 4. 与密钥相加: _m_paddb(result, key)

    解密过程（逆向）：
    1. 与密钥相减（逆向相加）
    2. 循环右移1位（逆向左移）
    3. 字节交换（逆向）
    4. 与密钥XOR（逆向）
    """
    # 将8字节转换为64位整数
    encrypted_value = struct.unpack("<Q", encrypted_block)[0]
    key_value = struct.unpack("<Q", key)[0]

    # 步骤1: 与密钥相减（逆向的_m_paddb）
    # _m_paddb是字节级加法，需要逐字节处理
    temp = 0
    for i in range(8):
        encrypted_byte = (encrypted_value >> (i * 8)) & 0xFF
        key_byte = (key_value >> (i * 8)) & 0xFF
        # 逆向字节加法（模256减法）
        decrypted_byte = (encrypted_byte - key_byte) & 0xFF
        temp |= decrypted_byte << (i * 8)

    # 步骤2: 循环右移1位（逆向的循环左移1位）
    temp = ((temp >> 1) | (temp << 63)) & 0xFFFFFFFFFFFFFFFF

    # 步骤3: 字节交换（按16位lane内交换高低字节）
    # 加密使用 _m_psrlwi/_m_psllwi 8 位并或，等价于对每个16位内交换字节；其逆操作同样是自身
    temp = (
        ((temp & 0x00FF00FF00FF00FF) << 8) | ((temp & 0xFF00FF00FF00FF00) >> 8)
    ) & 0xFFFFFFFFFFFFFFFF

    # 步骤4: 与密钥XOR（逆向）
    decrypted_value = temp ^ key_value

    # 转换回字节
    return struct.pack("<Q", decrypted_value)


def decrypt_file(encrypted_file, output_file, key_file=None):
    """
    解密文件

    Args:
        encrypted_file: 加密文件路径
        output_file: 输出文件路径
        key_file: 密钥文件路径（可选）
    """
    # 读取加密文件
    with open(encrypted_file, "rb") as f:
        encrypted_data = f.read()

    # 获取密钥
    if key_file and os.path.exists(key_file):
        with open(key_file, "rb") as f:
            key_data = f.read(8)  # 只读取前8字节
    else:
        # 使用默认密钥
        key_data = b"MMXEncode2024"[:8]

    # 解密数据
    decrypted_data = decrypt_data(encrypted_data, key_data)

    # 说明：加密程序写入的输出长度即为原始长度，未包含末尾2字节大小信息

    # 保存解密后的文件
    with open(output_file, "wb") as f:
        f.write(decrypted_data)

    print(f"解密完成: {encrypted_file} -> {output_file}")
    print(f"原始大小: {len(decrypted_data)} 字节")


def main():
    """主函数"""
    if len(sys.argv) < 3:
        print("用法: python decrypt.py <加密文件> <输出文件> [密钥文件]")
        print("示例: python decrypt.py encoded.dat decrypted.txt encoded.key")
        sys.exit(1)

    encrypted_file = sys.argv[1]
    output_file = sys.argv[2]
    key_file = sys.argv[3] if len(sys.argv) > 3 else None

    if not os.path.exists(encrypted_file):
        print(f"错误: 加密文件 '{encrypted_file}' 不存在")
        sys.exit(1)

    try:
        decrypt_file(encrypted_file, output_file, key_file)
    except Exception as e:
        print(f"解密失败: {e}")
        sys.exit(1)


if __name__ == "__main__":
    main()

```
